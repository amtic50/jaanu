IMPORTANT LESSON (11 Nov 2026): the first version of these rules (v3,
published ~14:48 IST) used `request.resource.data.<field> == null` as a
"checked only when present" guard. That pattern is WRONG: dot-access to a
field that is ABSENT from the written document raises
"Property <field> is undefined on object" and the whole rule errors out,
so the request is DENIED. Proven by the Rules Playground on 11 Nov 2026:
a create of jaanu_meta/shared with only {hist} failed at the `event`
check; a create with only {event} failed at `rewards`. Consequence: the
jaanu_meta/shared document was NEVER created — every meta write the app
attempted (events, rewards, freeze, love, why, pinHash) failed silently
(the app swallows cloud errors), so all shared settings lived in
localStorage on one phone only and never synced. Day check-ins were
unaffected because day writes always send the full day object.
CORRECT pattern: `request.resource.data.get('<field>', null)` — .get()
with a default returns null for absent fields instead of erroring.
All rules below use .get().
SECOND LESSON (same day): Firestore rules have NO per-item list
iteration — `.all(h, ...)` does not exist in the rules language (the
Playground reports "Function not found: [all]"), so a list field like
`hist` can only be validated as `is list` with a size cap; entries
cannot be checked one by one. Do NOT re-add `.all()`.

Console steps: https://console.firebase.google.com -> project "Jaanu"
(jaanu-ac156) -> Build -> Firestore Database -> Rules tab -> paste the
WHOLE block below -> Publish. Then verify in the Rules Playground:
create jaanu_meta/shared with {"hist": [{"start": "2026-05-09",
"days": 4}]} must be ALLOWED, and create with {"event": {"from":
"Amit", "type": "cheer", "text": "test", "ts": 1234567890}} must be
ALLOWED.

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ---- Jaanu (Prity's 3-month plan tracker) ----
    // Collection: jaanu_days — one document per day, id = YYYY-MM-DD.
    // Auth: Anonymous sign-in. Field validation uses .get(field, null)
    // so a partial write (missing fields) does not error the rule.
    match /jaanu_days/{date} {
      allow read: if request.auth != null;

      allow create, update: if request.auth != null
        && date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && request.resource.data.get('date', null) == date
        && (request.resource.data.get('gutka', null) == null
            || (request.resource.data.get('gutka', null) is int
                && request.resource.data.get('gutka', null) >= 0
                && request.resource.data.get('gutka', null) <= 6))
        && (request.resource.data.get('gum', null) == null
            || (request.resource.data.get('gum', null) is int
                && request.resource.data.get('gum', null) >= 0
                && request.resource.data.get('gum', null) <= 24))
        && (request.resource.data.get('alcohol', null) == null
            || request.resource.data.get('alcohol', null) is bool)
        && (request.resource.data.get('walk', null) == null
            || request.resource.data.get('walk', null) is bool)
        && (request.resource.data.get('walkMin', null) == null
            || (request.resource.data.get('walkMin', null) is int
                && request.resource.data.get('walkMin', null) >= 0
                && request.resource.data.get('walkMin', null) <= 240))
        && (request.resource.data.get('water', null) == null
            || (request.resource.data.get('water', null) is int
                && request.resource.data.get('water', null) >= 0
                && request.resource.data.get('water', null) <= 20))
        && (request.resource.data.get('sleep', null) == null
            || (request.resource.data.get('sleep', null) is number
                && request.resource.data.get('sleep', null) >= 0
                && request.resource.data.get('sleep', null) <= 14))
        && (request.resource.data.get('food', null) == null
            || request.resource.data.get('food', null) is bool)
        && (request.resource.data.get('period', null) == null
            || request.resource.data.get('period', null) is bool)
        && (request.resource.data.get('mood', null) == null
            || request.resource.data.get('mood', null) is string)
        && (request.resource.data.get('note', null) == null
            || (request.resource.data.get('note', null) is string
                && request.resource.data.get('note', null).size() <= 120));
      allow delete: if false;
    }
    // ---- end Jaanu ----

    // ---- Jaanu v3: shared meta doc (events, rewards, streak freeze) ----
    // Collection: jaanu_meta — one document only, id = "shared".
    // The app writes (merged, one field at a time): event {from, type,
    // text, ts}, rewards {name: text}, freeze {YYYY-MM-DD: true},
    // love {level: text}, why (string), pinHash (64-char SHA-256 hex),
    // hist [{start: "YYYY-MM-DD", days: int 1-10}] and reports
    // [{date, title, value, ref, note} — lab & scan results, read-only
    // for the app; not validated by the rule (allowed as an extra field)].
    match /jaanu_meta/shared {
      allow read: if request.auth != null;

      allow create, update: if request.auth != null
        && (request.resource.data.get('event', null) == null
            || (request.resource.data.get('event', null) is map
                && request.resource.data.get('event', null).from is string
                && request.resource.data.get('event', null).type is string
                && request.resource.data.get('event', null).text is string
                && request.resource.data.get('event', null).ts is int))
        && (request.resource.data.get('rewards', null) == null
            || request.resource.data.get('rewards', null) is map)
        && (request.resource.data.get('freeze', null) == null
            || request.resource.data.get('freeze', null) is map)
        && (request.resource.data.get('love', null) == null
            || request.resource.data.get('love', null) is map)
        && (request.resource.data.get('why', null) == null
            || request.resource.data.get('why', null) is string)
        && (request.resource.data.get('pinHash', null) == null
            || (request.resource.data.get('pinHash', null) is string
                && request.resource.data.get('pinHash', null).size() == 64))
        && (request.resource.data.get('hist', null) == null
            || (request.resource.data.get('hist', null) is list
                && request.resource.data.get('hist', null).size() <= 20));

      allow delete: if false;
    }
    // ---- end Jaanu v3 meta ----

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
